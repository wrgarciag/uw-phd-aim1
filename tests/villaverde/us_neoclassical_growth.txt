var c k y i;
varexo l n g;

parameters beta theta delta;

beta = 0.944;  // Discount factor to match 7.7% return to capital
theta = 0.39;  // Capital share
delta = 0.04;  // Depreciation rate

model;
// Production function (normalized)
y = k^theta * l^(1-theta);

// Resource constraint
c + i = y;

// Capital accumulation
i + (1-delta)*k(-1) = (1+g)*(1+n)*k;

// Euler equation
c^(-1) * (1+g) = beta * c(+1)^(-1) * (theta * k^(theta-1) * l(+1)^(1-theta) + 1-delta);
end;

initval;
k = ((1/beta - (1-delta))/theta)^(1/(theta-1)); // Steady-state capital (normalized)
y = k^theta; // Steady-state output (assuming l=1 for initial guess)
c = y - delta*k; // Steady-state consumption
i = delta*k; // Steady-state investment
l = 0.67; // Initial working-age ratio (approximate, based on U.S. data)
n = 0.0094; // Population growth rate (0.94% from Table 1)
g = 0.0165; // Technology growth rate for U.S. (Table 4)
end;

steady;

endval;
k = ((1/beta - (1-delta))/theta)^(1/(theta-1)); // Final steady-state capital
y = k^theta; // Final steady-state output (assuming l=1 for final guess)
c = y - delta*k; // Final steady-state consumption
i = delta*k; // Final steady-state investment
l = 0.65; // Final working-age ratio (approximate, adjusted for decline)
n = 0.0070; // Population growth rate in 2019 (Table 7, 2008-2019)
g = 0.0165; // Same technology growth rate
end;

// Exogenous variables evolution (approximate linear transition for l and n)
shocks;
var l;
periods 1:28;
values (linspace(0.67, 0.65, 28));
var n;
periods 1:28;
values (linspace(0.0094, 0.0070, 28));
var g;
periods 1:28;
values 0.0165;
end;

perfect_foresight_setup(periods=28);
perfect_foresight_solver;

rplot c k y i;
